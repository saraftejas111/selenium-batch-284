-- inner joins : returns only the matching data between 2 tables
SELECT student.name , student.subject , faculty.faculty_name
FROM student INNER JOIN faculty ON student.subject = faculty.lang ; 

-- left joins : returns all the data of left table and declare null if not matched in right table
SELECT student.name , student.subject , faculty.faculty_name
FROM student LEFT JOIN faculty ON student.subject = faculty.lang ; 


-- right joins : returns all the data of right table and declare null if not matched in left table
SELECT student.name , student.subject , faculty.faculty_name
FROM student RIGHT JOIN faculty ON student.subject = faculty.lang ;
